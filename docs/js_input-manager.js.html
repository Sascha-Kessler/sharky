/**
 * Manages all player input (keyboard, mouse, and touch).
 */
class InputManager {
  /**
   * Creates a new InputManager.
   * @param {Keyboard} keyboard - Shared keyboard state object
   */
  constructor(keyboard) {
    this.keyboard = keyboard;

    /**
     * Maps keyboard keys to internal actions
     * @type {Object<string, string>}
     */
    this.keyMap = {
      ArrowLeft: "LEFT",
      ArrowRight: "RIGHT",
      ArrowUp: "UP",
      ArrowDown: "DOWN",
      Space: "ATTACK",
      KeyB: "POISON",
      KeyV: "FIN_SLAP",
    };

    window.addEventListener("keydown", (e) => this.handleKey(e, true));
    window.addEventListener("keyup", (e) => this.handleKey(e, false));

    this.initTouchControls();

    window.addEventListener("pointerup", () => this.resetTouchKeys());
    window.addEventListener("pointercancel", () => this.resetTouchKeys());
    window.addEventListener("blur", () => this.resetTouchKeys());
  }

  /**
   * Handles keyboard input and updates the key state.
   * @param {KeyboardEvent} event - Keyboard event
   * @param {boolean} isPressed - Whether the key is pressed or released
   */
  handleKey(event, isPressed) {
    const key = this.keyMap[event.code];
    if (key) {
      this.keyboard[key] = isPressed;
    }
  }

  /**
   * Initializes touch and mouse controls for mobile buttons.
   */
  initTouchControls() {
    document.querySelectorAll(".touch-btn").forEach((btn) => {
      const key = btn.dataset.key;
      if (!key) return;

      this.addTouchButtonEvents(btn, key);
    });

    this.addGlobalResetEvents();
  }

  /**
   * Adds all relevant input event listeners to a touch button.
   * @param {HTMLElement} btn - The button element
   * @param {string} key - The mapped keyboard action
   */
  addTouchButtonEvents(btn, key) {
    btn.addEventListener("touchstart", (e) => this.handleTouchStart(e, key), {
      passive: false,
    });

    btn.addEventListener("touchend", () => this.setKey(key, false));
    btn.addEventListener("touchcancel", () => this.setKey(key, false));
    btn.addEventListener("mousedown", () => this.setKey(key, true));
    btn.addEventListener("mouseup", () => this.setKey(key, false));
    btn.addEventListener("mouseleave", () => this.setKey(key, false));
  }

  /**
   * Handles touchstart event and prevents default browser behavior.
   * @param {TouchEvent} event - Touch event
   * @param {string} key - The mapped keyboard action
   */
  handleTouchStart(event, key) {
    if (event.cancelable) {
      event.preventDefault();
    }

    this.setKey(key, true);
  }

  /**
   * Sets the state of a specific key.
   * @param {string} key - The key identifier
   * @param {boolean} value - The key state (pressed or released)
   */
  setKey(key, value) {
    this.keyboard[key] = value;
  }

  /**
   * Adds global listeners to reset stuck keys on interaction end.
   */
  addGlobalResetEvents() {
    window.addEventListener("touchend", () => this.resetTouchKeys());
    window.addEventListener("mouseup", () => this.resetTouchKeys());
    window.addEventListener("blur", () => this.resetTouchKeys());
  }

  /**
   * Resets all input states to prevent stuck keys.
   */
  resetTouchKeys() {
    this.keyboard.LEFT = false;
    this.keyboard.RIGHT = false;
    this.keyboard.UP = false;
    this.keyboard.DOWN = false;
    this.keyboard.ATTACK = false;
    this.keyboard.POISON = false;
    this.keyboard.FIN_SLAP = false;
  }
}
